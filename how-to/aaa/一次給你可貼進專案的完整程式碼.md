## 5) HTTP 與 API 層

### `src/api/http.ts`

```ts
import axios from 'axios'

const http = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL || '/api',
  timeout: 20000
})

http.interceptors.request.use((config) => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

http.interceptors.response.use(
  (response) => response.data,
  (error) => {
    const status = error?.response?.status
    if (status === 401) {
      localStorage.removeItem('token')
      if (location.pathname !== '/login') location.href = '/login'
    }
    return Promise.reject(error)
  }
)

export default http
```


### `src/api/asset.ts`

```ts
import http from './http'
import type { ApiResponse } from '@/types/common'
import type { AssetInventoryRow, AssetQuery, InventoryDateRow, ScrappedQuery } from '@/types/asset'

const prefix = '/asset'

export const assetApi = {
  getInventoryDate: () =>
    http.post<any, ApiResponse<InventoryDateRow[]>>(`${prefix}/getInventoryDate`, {}),

  getAssetInventory: (payload: AssetQuery) =>
    http.post<any, ApiResponse<AssetInventoryRow[]>>(`${prefix}/getAssetInventory`, payload),

  getStaffAsset: (payload: { sidx?: string; sord?: string } = {}) =>
    http.post<any, ApiResponse<AssetInventoryRow[]>>(`${prefix}/getStaffAsset`, payload),

  getAssetScrapped: (payload: ScrappedQuery) =>
    http.post<any, ApiResponse<AssetInventoryRow[]>>(`${prefix}/getAssetScrapped`, payload),

  uploadExcel: (file: File) => {
    const form = new FormData()
    form.append('file', file)
    return http.post<any, ApiResponse<AssetInventoryRow[]>>(`${prefix}/uploadExcel`, form, {
      headers: { 'Content-Type': 'multipart/form-data' }
    })
  },

  saveImport: (payload: { assetinventorydata: AssetInventoryRow[] }) =>
    http.post<any, ApiResponse>(`${prefix}/saveImport`, payload),

  saveAsset: (payload: { inventorydate: string; assetinventorydata: AssetInventoryRow[] }) =>
    http.post<any, ApiResponse>(`${prefix}/saveAsset`, payload),

  deleteAll: (inventorydate: string) =>
    http.post<any, ApiResponse>(`${prefix}/deleteAll`, { inventorydate }),

  deleteOne: (inventorydate: string, selfno: string) =>
    http.post<any, ApiResponse>(`${prefix}/delete`, { inventorydate, selfno }),

  scrapped: (inventorydate: string, selfno: string) =>
    http.post<any, ApiResponse>(`${prefix}/scrapped`, { inventorydate, selfno }),

  newAsset: () =>
    http.post<any, ApiResponse>(`${prefix}/newAsset`, {}),

  submit: () =>
    http.post<any, ApiResponse>(`${prefix}/submit`, {}),

  inform: () =>
    http.post<any, ApiResponse>(`${prefix}/inform`, {}),

  getAssetTypeHtml: () =>
    http.get<string>(`${prefix}/getAssetType`, { responseType: 'text' })
}
```

```ts
import { createRouter, createWebHistory } from 'vue-router'
import { routes } from './routes'

const router = createRouter({
  history: createWebHistory(),
  routes
})

router.beforeEach((to) => {
  const token = localStorage.getItem('token')
  if (!to.meta.public && !token && to.path !== '/login') return '/login'
})

export default router
```
