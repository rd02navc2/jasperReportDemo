CREATE TABLE `erp_coupon_log` (

  `center_id` varchar(20) NOT NULL COMMENT '據點/商場代號 (sCenter)',

  `card_no` varchar(50) DEFAULT NULL COMMENT '會員卡號 (sCardNO)',
  `coupon_no` varchar(50) NOT NULL COMMENT '折價券號 (lCouponNO/sCouponNO)',
  `pos_id` varchar(20) DEFAULT NULL COMMENT 'POS機台號 (sPOSID)',
  `counter_id` varchar(20) DEFAULT NULL COMMENT '櫃位代號 (sCounterID)',
  `trade_date` date NOT NULL COMMENT '交易日期 (sDate)',
  `amount` decimal(10,2) DEFAULT '0.00' COMMENT '金額 (iAmt)',
  `status` varchar(10) DEFAULT 'N' COMMENT '狀態: N-未使用, Y-已使用, E-已過期',
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP COMMENT '建立時間',
  `updated_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新時間',
  PRIMARY KEY (`id`),
  KEY `idx_coupon_center` (`coupon_no`, `center_id`),
  KEY `idx_card_no` (`card_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='ERP折價券核銷紀錄表';

驗證核銷結果的 SQL 指令

當你用 Postman 發送請求並得到成功回應後，可以用這段指令來模擬或確認 ERP 資料庫端是否正確更新了狀態與銷貨資訊：
SQL

-- 模擬 ERP 接收到請求後更新狀態
UPDATE erp_coupon_log 
SET status = 'Y', 
    sale_no = 'TG-202607199999', 
    pos_id = 'POS02', 
    counter_id = 'CT008'
WHERE center_id = 'BY001' 
  AND coupon_no IN ('A20260719001', 'A20260719002');

-- 查詢確認更新是否成功
SELECT coupon_no, center_id, sale_no, status, updated_at 
FROM erp_coupon_log 
WHERE coupon_no IN ('A20260719001', 'A20260719002');

  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '自增主鍵',
  `center_id` varchar(20) NOT NULL COMMENT '商場代號 (sCenter)',
