CREATE TABLE `line_push_log` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '流水號',
  `report_type` VARCHAR(5) NOT NULL COMMENT '報表類型(A/B/C/D)',
  `report_date` DATE NOT NULL COMMENT '營業日期',
  `file_name` VARCHAR(255) NOT NULL COMMENT '圖檔名稱',
  `public_url` VARCHAR(500) DEFAULT NULL COMMENT '對外公開的圖片網址',
  `target_id` VARCHAR(100) NOT NULL COMMENT 'LINE 接收端 ID',
  `status` VARCHAR(20) NOT NULL COMMENT '狀態: SUCCESS / FAILED',
  `error_message` TEXT COMMENT '失敗原因的 JSON 或錯誤訊息',
  `created_time` TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '建立時間',
  PRIMARY KEY (`id`),
  KEY `idx_report_date` (`report_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='LINE 營業報表推播紀錄表';


Java 內執行的預期 SQL Insert 指令

當你的 Feign 呼叫成功或失敗後，可以在 Service 層透過 Spring Data JPA 或 JdbcTemplate 寫入資料庫：
成功發送時的 Insert 範例：
SQL

INSERT INTO line_push_log (report_type, report_date, file_name, public_url, target_id, status, error_message)
VALUES (
  'B', 
  '2026-07-19', 
  'revenue_report_20260719.png', 
  'https://yourdomain.com/static/revenue/revenue_report_20260719.png', 
  'Cxxxxxxxxxxxxxxx_BC', 
  'SUCCESS', 
  NULL
);

失敗發送時的 Insert 範例（紀錄 FeignException 的錯誤回傳）：
SQL
