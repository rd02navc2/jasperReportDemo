--- 1. BonusController.java (路徑 A 入口) ---
@PostMapping("/addPoint")
public BonusDTO.Response<Void> addPoint(@RequestBody BonusDTO.Request req) {
    try {
        // 智慧分流判斷
        if ((req.getOrderNo() != null && !req.getOrderNo().isEmpty()) || isVipPromo(req.getLrq01())) {
            String billNo = (req.getOrderNo() != null) ? "VIP_GIFT_" + req.getOrderNo() : "POS_VIP_" + System.currentTimeMillis();
            bonusService.addVipGiftPoint(req.getCenter(), req.getCounterId(), req.getCardNo(), req.getPoint(), billNo);
        } else {
            String lrq01 = (req.getLrq01() != null) ? req.getLrq01() : "603";
            String lrq02 = (req.getLrq02() != null) ? req.getLrq02() : "603";
            bonusService.addPoint(req.getCenter(), req.getCounterId(), req.getCardNo(), lrq01, lrq02, req.getPoint());
        }
        return BonusDTO.Response.success(null);
    } catch (Exception e) {
        throw new ResponseStatusException(HttpStatus.BAD_REQUEST, e.getMessage());
    }
}

--- 2. BonusService.java (核心邏輯封裝) ---
@Transactional(rollbackFor = Exception.class)
public void addVipGiftPoint(String center, String counterId, String cardNo, int points, String accessId) {
    // 冪等性保障
    if (bonusLogRepository.existsByAccessId(accessId)) return;

    // 寫入日誌 (Audit)
    bonusLogRepository.save(BonusLog.builder().center(center).counterId(counterId).accessId(accessId).point(points).build());

    // 寫入流水帳 (LSM_FILE)
    LsmFile lsm = new LsmFile();
    lsm.setLsm01(cardNo);
    lsm.setLsm03(accessId); // 將字軌設為關聯單號
    lsm.setLsm04((double)points);
    lsmFileRepository.save(lsm);

    // 發送事件
    boolean ok = streamBridge.send("bonus-out-0", PointChangedEvent.builder().billNo(accessId).build());
    if (!ok) throw new RuntimeException("Kafka 發送失敗");
}
